cmake_minimum_required(VERSION 3.28)
project(AwesomeEngine CXX)

set(SFML_BUILD_NETWORK ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SPDLOG_USE_STD_FORMAT ON)
set(TGUI_BACKEND SFML_GRAPHICS)
set(TGUI_BUILD_GUI_BUILDER OFF)

include(FetchContent)
FetchContent_Declare(sfml GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 3.1.0)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz)
FetchContent_Declare(logs URL https://github.com/gabime/spdlog/archive/refs/tags/v1.16.0.tar.gz)
FetchContent_Declare(tgui URL https://github.com/texus/TGUI/archive/refs/tags/v1.13.0.tar.gz)
FetchContent_Declare(magic_enum URL https://github.com/Neargye/magic_enum/archive/refs/tags/v0.9.8.tar.gz)
FetchContent_MakeAvailable(sfml json logs tgui magic_enum)

file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS Engine/Source/*.cpp Games/Source/*.cpp)
add_executable(AwesomeEngine ${SOURCES})

target_include_directories(AwesomeEngine PRIVATE Engine/Include Games/Include)
target_compile_features(AwesomeEngine PRIVATE cxx_std_20)
target_compile_options(AwesomeEngine PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_compile_options(AwesomeEngine PRIVATE $<IF:$<CXX_COMPILER_ID:MSVC>, /W4, -Wall -Wextra>)
target_link_libraries(AwesomeEngine PRIVATE SFML::Graphics SFML::Audio SFML::Network nlohmann_json::nlohmann_json spdlog::spdlog TGUI::TGUI magic_enum::magic_enum)

include(CTest)
if(BUILD_TESTING)
  add_executable(NoTraceNetworkTests Tests/NoTraceNetworkTests.cpp Engine/Source/Managers/NetworkManager.cpp)
  target_include_directories(NoTraceNetworkTests PRIVATE Engine/Include)
  target_compile_features(NoTraceNetworkTests PRIVATE cxx_std_20)
  target_compile_options(NoTraceNetworkTests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  target_link_libraries(NoTraceNetworkTests PRIVATE SFML::Network)
  add_test(NAME NoTraceNetworkTransport COMMAND NoTraceNetworkTests)
  set_tests_properties(NoTraceNetworkTransport PROPERTIES TIMEOUT 30)
  add_executable(NoTraceSimulationTests Tests/NoTraceSimulationTests.cpp Games/Source/NoTraceSimulation.cpp Games/Source/NoTraceMission.cpp)
  target_include_directories(NoTraceSimulationTests PRIVATE Games/Include)
  target_compile_features(NoTraceSimulationTests PRIVATE cxx_std_20)
  target_compile_options(NoTraceSimulationTests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  target_link_libraries(NoTraceSimulationTests PRIVATE SFML::Graphics nlohmann_json::nlohmann_json)
  add_test(NAME NoTraceSimulation COMMAND NoTraceSimulationTests)
  set_tests_properties(NoTraceSimulation PROPERTIES TIMEOUT 30)
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND WIN32)
  set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,--start-group")
  set(CMAKE_CXX_STANDARD_LIBRARIES "${CMAKE_CXX_STANDARD_LIBRARIES} -Wl,--end-group")
endif()
